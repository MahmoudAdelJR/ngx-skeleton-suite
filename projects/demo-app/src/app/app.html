INSTALLATION & SETUP

GETTING STARTED WITH NGX-SKELETON

Enterprise structural directive suite for Angular 18+ and PrimeNG.

01. NPM INSTALLATION

Install the package via npm or yarn in your Angular workspace:

npm install ngx-skeleton-suite

02. STANDALONE COMPONENT USAGE (RECOMMENDED)

Import SkeletonDirective directly into your standalone component:

import { Component, signal } from '@angular/core';
import { SkeletonDirective } from 'ngx-skeleton-suite';

@Component({
  selector: 'app-user-profile',
  standalone: true,
  imports: [SkeletonDirective],
  template: `
    <div *skeleton="isLoading(); height: '12rem'; animation: 'wave'">
      <app-user-card [user]="user()" />
    </div>
  `
})
export class UserProfileComponent {
  isLoading = signal(true);
}

03. GLOBAL PROVIDER CONFIGURATION

Configure default animation modes, delay timers, and custom preset registries in app.config.ts:

import { ApplicationConfig } from '@angular/core';
import { provideNgxSkeleton } from 'ngx-skeleton-suite';

export const appConfig: ApplicationConfig = {
  providers: [
    provideNgxSkeleton({
      animation: 'wave',
      delayMs: 80,
      minimumDurationMs: 200,
      fadeInMs: 180,
      preserveLayout: true,
    }),
  ],
};
